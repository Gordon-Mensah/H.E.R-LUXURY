## Step 5 — Paste the Design Panel JavaScript into admin.html

Add this block inside the existing `<script>` tag at the bottom of admin.html (before the closing `</script>`):

```javascript
// ── DESIGN STUDIO ─────────────────────────────────────────────

let currentTheme = {};
let previewDevice = 'desktop';

const COLOR_CONTROLS = [
  { label: 'Accent (Rose/CTA)',    key: '--rose' },
  { label: 'Accent Dark',          key: '--rose-dark' },
  { label: 'Gold / Price',         key: '--gold' },
  { label: 'Gold Light',           key: '--gold-light' },
  { label: 'Page Background',      key: '--ivory' },
  { label: 'Surface / Cards',      key: '--surface' },
  { label: 'Card Background',      key: '--card' },
  { label: 'Cream Sections',       key: '--cream' },
  { label: 'Parchment Sections',   key: '--parchment' },
  { label: 'Blush Tones',          key: '--blush' },
  { label: 'Primary Text',         key: '--text' },
  { label: 'Secondary Text',       key: '--text-mid' },
  { label: 'Muted Text',           key: '--text-muted' },
  { label: 'Border Color',         key: '--border' },
  { label: 'Taupe / Accents',      key: '--taupe' },
];

function initDesignPanel() {
  currentTheme = loadDesignTheme();
  renderColorControls();
  renderPresetBtns();
  syncUIFromTheme();
  applyThemeToPreview();
}

function renderColorControls() {
  const wrap = document.getElementById('colorControls');
  if (!wrap) return;
  wrap.innerHTML = COLOR_CONTROLS.map(({ label, key }) => {
    const val = currentTheme[key] || '#000000';
    return `
      
      
        ${label}
        ${val}
      
    `;
  }).join('');
}

function renderPresetBtns() {
  const wrap = document.getElementById('presetBtns');
  if (!wrap) return;
  wrap.innerHTML = Object.keys(PRESETS).map(name =>
    `
      ${name}
    `
  ).join('');
}

function syncUIFromTheme() {
  // Colors
  COLOR_CONTROLS.forEach(({ key }) => {
    const input = document.querySelector(`[data-key="${key}"]`);
    if (input) input.value = currentTheme[key] || '#000000';
    const valEl = document.getElementById('colorVal' + key.replace(/--/g,'').replace(/-/g,'_'));
    if (valEl) valEl.textContent = currentTheme[key] || '';
  });

  // Fonts
  const fh = document.getElementById('fontHeading');
  if (fh) { [...fh.options].forEach(o => { if (currentTheme['--font-heading'] && currentTheme['--font-heading'].includes(o.value.split(',')[0].replace(/'/g,''))) o.selected = true; }); fh.value = currentTheme['--font-heading'] || fh.value; }
  const fb = document.getElementById('fontBody');
  if (fb) { fb.value = currentTheme['--font-body'] || fb.value; }

  // Text
  const texts = [
    ['txtStoreName',    '--store-name'],
    ['txtHeroEyebrow',  '--hero-eyebrow'],
    ['txtHeroTitle',    '--hero-title-text'],
    ['txtHeroSubtitle', '--hero-subtitle-text'],
    ['txtMarquee',      '--marquee-text'],
    ['txtFooter',       '--footer-copy'],
  ];
  texts.forEach(([id, key]) => {
    const el = document.getElementById(id);
    if (el) el.value = currentTheme[key] || '';
  });

  // Card radius
  const r = parseInt(currentTheme['--card-radius'] || '0');
  const slider = document.getElementById('cardRadius');
  if (slider) slider.value = r;
  const rVal = document.getElementById('cardRadiusVal');
  if (rVal) rVal.textContent = r + 'px';

  // Section toggles
  const toggles = [
    ['showMarquee',      '--show-marquee'],
    ['showMaterials',    '--show-materials'],
    ['showTestimonials', '--show-testimonials'],
    ['showAbout',        '--show-about'],
  ];
  toggles.forEach(([id, key]) => {
    const el = document.getElementById(id);
    if (el) el.checked = (currentTheme[key] !== '0');
  });
}

function applyThemeToPreview() {
  saveDesignTheme(currentTheme);
  const iframe = document.getElementById('designPreview');
  if (!iframe) return;
  try {
    const doc = iframe.contentDocument || iframe.contentWindow.document;
    let style = doc.getElementById('__designer_overrides__');
    if (!style) {
      style = doc.createElement('style');
      style.id = '__designer_overrides__';
      doc.head.appendChild(style);
    }
    const vars = Object.entries(currentTheme)
      .filter(([k]) => k.startsWith('--') && !k.startsWith('--show') && !k.includes('text-') && !k.includes('font-') && !k.includes('hero') && !k.includes('store') && !k.includes('marquee') && !k.includes('footer') && !k.includes('eyebrow') && !k.includes('radius'))
      .map(([k, v]) => `${k}:${v};`).join('');
    style.textContent = `:root{${vars}}`;

    // Also update text + layout vars
    const root = doc.documentElement;
    Object.entries(currentTheme).forEach(([k, v]) => {
      if (k.startsWith('--')) root.style.setProperty(k, v);
    });
  } catch(e) {
    // Cross-origin or not ready — just save and let user refresh
  }
}

function refreshPreview() {
  const iframe = document.getElementById('designPreview');
  if (!iframe) return;
  saveDesignTheme(currentTheme);
  iframe.src = 'index.html?' + Date.now();
  setTimeout(applyThemeToPreview, 1200);
}

function setPreviewDevice(device) {
  previewDevice = device;
  const iframe = document.getElementById('designPreview');
  const wrap = document.getElementById('previewWrap');
  const desktopBtn = document.getElementById('prevDesktop');
  const mobileBtn  = document.getElementById('prevMobile');
  if (device === 'mobile') {
    iframe.style.width = '390px';
    iframe.style.maxWidth = '390px';
    wrap.style.padding = '1.5rem';
    desktopBtn.style.background = 'none';
    desktopBtn.style.color = 'var(--text-dim)';
    mobileBtn.style.background = 'var(--text)';
    mobileBtn.style.color = 'var(--white)';
  } else {
    iframe.style.width = '100%';
    iframe.style.maxWidth = '1200px';
    wrap.style.padding = '1rem';
    desktopBtn.style.background = 'var(--text)';
    desktopBtn.style.color = 'var(--white)';
    mobileBtn.style.background = 'none';
    mobileBtn.style.color = 'var(--text-dim)';
  }
}

function onColorChange(key, value) {
  pushHistory(currentTheme);
  currentTheme[key] = value;
  const valEl = document.getElementById('colorVal' + key.replace(/--/g,'').replace(/-/g,'_'));
  if (valEl) valEl.textContent = value;
  applyThemeToPreview();
}

function onFontChange(key, value) {
  pushHistory(currentTheme);
  currentTheme[key] = value;
  applyThemeToPreview();
}

function onTextChange(key, value) {
  currentTheme[key] = value;
  applyThemeToPreview();
}

function onRadiusChange(value) {
  currentTheme['--card-radius'] = value + 'px';
  document.getElementById('cardRadiusVal').textContent = value + 'px';
  applyThemeToPreview();
}

function onSectionToggle(key, checked) {
  pushHistory(currentTheme);
  currentTheme[key] = checked ? '1' : '0';
  applyThemeToPreview();
}

function applyPreset(name) {
  pushHistory(currentTheme);
  currentTheme = { ...PRESETS[name] };
  syncUIFromTheme();
  applyThemeToPreview();
  showToast('Applied ' + name + ' theme');
}

function saveAndPreview() {
  saveDesignTheme(currentTheme);
  refreshPreview();
  showToast('Design saved! Storefront updated ✓');
}

function undoDesign() {
  const prev = undoHistory();
  if (!prev) { showToast('Nothing to undo', true); return; }
  currentTheme = prev;
  syncUIFromTheme();
  applyThemeToPreview();
  showToast('Undone');
}

function resetDesign() {
  if (!confirm('Reset all design changes to default? This cannot be undone.')) return;
  localStorage.removeItem('her_design_history');
  currentTheme = resetDesignTheme();
  syncUIFromTheme();
  applyThemeToPreview();
  refreshPreview();
  showToast('Reset to default design');
}

function copyThemeCSS() {
  const css = exportThemeAsCSS(currentTheme);
  navigator.clipboard.writeText(css).then(() => {
    showToast('CSS copied to clipboard! Paste into :root in style.css');
  }).catch(() => {
    showToast('Could not copy — try manually', true);
    console.log(css);
  });
}

function setDesignInput(text) {
  document.getElementById('designAiInput').value = text;
  document.getElementById('designAiInput').focus();
}

async function sendDesignCommand() {
  const input = document.getElementById('designAiInput');
  const sendBtn = document.getElementById('designSendBtn');
  const message = input.value.trim();
  if (!message) return;

  const settings = getSettings();
  if (!settings.groqApiKey) {
    showToast('Add your Groq API key in Settings first', true);
    showPanel('settings');
    return;
  }

  input.value = '';
  sendBtn.disabled = true;
  addMessage('designAiMessages', 'user', message);
  addLoadingMessage('designAiMessages');

  try {
    const result = await processDesignCommand(message, currentTheme, settings.groqApiKey);
    removeLoadingMessage();

    if (result.changes && Object.keys(result.changes).length > 0) {
      pushHistory({ ...currentTheme });
      currentTheme = { ...currentTheme, ...result.changes };
      syncUIFromTheme();
      applyThemeToPreview();
      addMessage('designAiMessages', 'ai', result.message + 'Changed: ' + Object.keys(result.changes).join(', ') + '', 'success');
    } else {
      addMessage('designAiMessages', 'ai', result.message);
    }
  } catch(err) {
    removeLoadingMessage();
    addMessage('designAiMessages', 'ai', 'Error: ' + err.message, 'error');
    showToast('AI error', true);
  }
  sendBtn.disabled = false;
}
```

## Step 6 — Add theme loader to index.html (and any other storefront pages)

```html
(function(){
  try {
    var t = localStorage.getItem('her_design_theme');
    if (!t) return;
    var theme = JSON.parse(t);
    var root = document.documentElement;
    Object.entries(theme).forEach(function(e) {
      if (e[0].startsWith('--')) root.style.setProperty(e[0], e[1]);
    });
  } catch(e) {}
})();
```

```html
(function(){
  try {
    var t = localStorage.getItem('her_design_theme');
    if (!t) return;
    var theme = JSON.parse(t);
    
    // Handle font loading
    if (theme['--font-heading'] || theme['--font-body']) {
      var fonts = [];
      var fh = theme['--font-heading'] || '';
      var fb = theme['--font-body'] || '';
      var extractFont = function(s) { var m = s.match(/'([^']+)'/); return m ? m[1] : ''; };
```

## How it works end-to-end

1. Admin opens Design Studio panel
2. They use AI chat or manual controls to change colors/fonts/text
3. Changes are saved to `localStorage` under key `her_design_theme`
4. The live preview iframe reloads `index.html` — which reads from `localStorage` on load and applies all CSS variables before first paint
5. When they click "Save & Apply", `saveDesignTheme()` writes the final theme
6. Every visitor to the storefront gets the custom theme automatically
7. "Copy CSS" exports the diff as a proper `:root {}` block they can paste into `style.css` to make changes permanent
